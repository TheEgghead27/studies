%\section{Foundations: Signals and Sequences}
\section{Signals}
A \textbf{signal} conveys information as a function over some independent variable(s). The independent variable tells us what it means for two values to be adjacent, delayed, shifted, or repeated.

A \textbf{continuous-time (CT)} signal $x_c(t)$ has a continuous independent variable $t\in\mathbb{R}$ and continuous amplitudes.

A \textbf{discrete-time (DT)} signal $x[n]$ has a discrete independent variable $n\in\mathbb{Z}$ and continuous or quantized amplitudes. A \textbf{digital} signal $x_q[n]$ is a special case of the DT signal where the amplitude is restricted to a discrete, finite set of levels.

\subsection{Sampling}
For a sampling period $T$ (sampling frequency $f_s=\frac{1}{T}$)
\begin{equation}
    x[n]=x_a(nT)
\end{equation}


A DT system $\textit{T}$ maps an input sequence to an output sequence.
\begin{equation}
    y[n] = \textit{T}\{x[n]\}
\end{equation}
This takes into account ALL information in the input sequence ($-\infty<n<\infty$), not necessarily just the present $n$ (i.e. a memoryless system). We refer to $x[n]$ as the $n^{th}$ sample even if the signal was not generated by sampling.

\subsection{The Unit Sample Impulse}
\[
    \delta[n] = \begin{cases}
    1 & n=0\\
    0 & n\ne0
    \end{cases}
\]
A shifted impulse $\delta[n-n_0]$ will move the spike to $n=n_0$. Any DT sequence can be represented as a linear combination of shifted deltas
\[
    x[n] = \sum_{k=-\infty}^\infty x[k]\delta[n-k]
\]
Each delayed inpulse selects an index and the scale supplies the value at that index. Delayed impulses are a coordinate basis for sequences.

\subsection{The Unit Step Sequence}

\[
    u[n] = \begin{cases}
    1 & n\ge0\\
    0 & n<0
    \end{cases} = \sum_{k=0}^\infty \delta[n-k]
\]
Notice that $\delta[n] = u[n] - u[n-1]$ (a backward difference, analogous to differentiation).

One can define a finite-duration rectangular window sequence from $n=0$ to $n=N-1$ as $$u_N[n] = u[n]  - u[n-N]$$.

\subsection{Exponential Sequences}
\[
    x[n] - A\alpha^n
\]
In the real case, we have $A,\alpha\in\mathbb{R}$ and the specific values yield properties
\begin{itemize}
	\item $A>0$, the sequence starts positive
	\item $|\alpha|<1$, we have a sequence that decreases in magnitude $|x[n = |A||\alpha|^n$ (the envelope shrinks)
	\item $|\alpha|>1$, we have a growing sequence
	\item $|\alpha|=1$, there is no decay.
	\item $\alpha<0$, we have alternating signs
\end{itemize}
Note that $\alpha = \frac{x[n+1]}{x[n]}$.

In the complex case, we have $\alpha=|\alpha|e^{j\omega_0}$, $A=|A|e^{j\phi}$, yielding
\begin{align}
    x[n] = A\alpha^n=&|A||\alpha|^ne^{j(\omega_0 n + \phi)}\\
                    =&|A||\alpha|^n(\cos(w_0n+\phi) + j\sin(w_0n+\phi))
\end{align}
the value of $A$ sets the initial magnitude/phase, $\omega_0$ sets the oscillation (radians per sample), and $|a|$ controls the envelope.
$w_0$ and $w_0+2\pi$ are indistinguishable in DT (the $2\pi$ term turns into $1$), we can consider $w_0$ near $0$ to be low frequency, and $w_0$ near $\pi$ to be high frequency (with any higher being alias frequencies).

A DT signal is periodic if $x[n+N] = x[n]$, which would require that $e^{j\omega_0n} = 1$, or equivalently $\omega_0N = 2\pi k$, so \[\frac{\omega_0}{2\pi} = \frac{k}{N}\in\mathbb{Q}\]
We can correlate $w_0$ with a frequency in (continuous) time $f_0$ with \[\omega_0=2\pi\frac{f_0}{f_s}\]

\section{Systems}
A system maps a signal $x:\mathbb{Z}\to\mathbb{C}$ to another signal.
\[y[n] = T\{x[n]\}\]
\subsection{Examples}
\subsubsection{Ideal Delay System}
For a fixed delay $n_d$
\[y[n] = x[n-n_d]\]
\subsubsection{General Moving-Average System}
For two indices $M_1,M_2\ge0\in\mathbb{Z}$
\begin{equation}
    y[n] = \frac{1}{M_1+M_2+1}\sum_{k=-M_1}^{M_2}x[n-k]
\end{equation}
This is used to smooth out signal noise when we expect  the noise to have 0 mean.

%\section{Foundations II: LTI Systems, Convolution, System Properties}
\subsection{Memoryless Systems}
A system is memoryless if and only if $\forall n,$ $y[n]$ depends only on $x[n]$ at the same $n$.
\subsubsection{The Ideal Delay System}
\begin{equation}
    y[n] = x[n-n_d]
\end{equation}
This system shifts the signal in the time domain by a fixed integer offset $n_0\in\mathbb{Z}$. $n_d>0$ shifts the signal to the right by $n_d$, $n_d<0$ shifts the signal to the left by $n_d$ (this is also called a time advance).

This requires memory for all nontrivial $n_d\ne0$.
\subsubsection{The Accumulator System}
\begin{equation}
    y[n]=\sum_{k=-\infty}^n x[k]
\end{equation}
This system accumulates (sums) all inputs at the present and past. This requires memory.

\subsection{Linear Systems}
A system is linear if and only if the following conditions hold
\begin{enumerate}
    \item Additivity -- $\textit{T}(x_1[n] + x_2[n]) = \textit{T}(x_1[n])+\textit{T}(x_2[n])$
    \item Scaling/Homogeneity -- $T(ax_1[n]) = aT(a[n])$
\end{enumerate}
It is sufficient to prove the combined statement
\[
    T(ax_1[n]+bx_2[n])=aT(x_1[n])+bT(x_2[n])
\]
and generalize it as
\begin{align*}
    x[n] &= \sum_k a_k[n] \\
    y[n] &= \sum_k a_ky_k[n]\\
    y_k[n] &= T(x_k[n])
\end{align*}


\subsubsection{The Accumulator System}
Through evaluation of the accumulator, we can find that it is linear.

\subsection{Time Invariant Systems}
A system is time-invariant if $\forall n_0\in\mathbb{Z}$ we have $x_1[n]=x[n-n_0]$ with corresponding $y_1[n]=y[n-n_0]$. Delaying a signal, then processing it has the same result as processing a signal, then delaying it.

\subsubsection{The Accumulator System}
The accumulator system is time invariant by use of a substitution variable for evaluation.

\subsubsection{The Compressor System}
For $M\in\mathbb{Z}_+$
\begin{align}
    y[n]=x[Mn] &&-\infty<n<\infty
\end{align}
Every $M^{th}$ sample is retained, with all others being discarded.
This system is linear, but it is not time invariant ($x[M(n-n_0)]\ne x[Mn-n_0]$).


\subsection{Causal Systems}
A system is causal if and only if $\forall n_0\in\mathbb{Z}, y[n_0]$ depends only on $x[n \le  n_0]$.

Equivalently, given $x_1[n]=x_2[n]$ for all $n\le n_0$, a causal system has $y_1[n]=y_n2[n]$ for all $n\le n_0$.

\subsubsection{Difference Systems}
The forward difference system is
\begin{equation}
    y[n] = x[n+1]-x[n]
\end{equation}
and the backward difference system is
\begin{equation}
    y[n] = x[n]-x[n-1]
\end{equation}

The latter is causal, but the former is not.

\subsection{BIBO Stable Systems}
A sequence $x[n]$ is bounded if there exists a fixed, finite, positive value $B_x$ such that
\begin{equation}
    |x[n]|\le B_x<\infty
\end{equation}

A system is bounded-input, bounded-output stable iff every bounded input sequence produces a bounded output sequence.

Existence of even a single bounded input resulting in unbounded output will disprove stability. To prove stability, we must show stability for all bounded inputs.

\subsubsection{The Accumulator System}
The accumulator is unstable, for a bounded input $x[n]=u[n]$ with bound $B_x=1$, the accumulated output is
\[y[n] = \begin{cases*}
n+1 &if $n\ge 0$\\
0 & otherwise
\end{cases*}
\]
which we cannot find a $B_y<\infty$ for.


\section{Introduction to LTI Systems}
LTI systems are both linear and time invariant. This is useful because any sequence can be written as a linear combination of delayed impulses
$$x[n] = \sum_{k=-\infty}^\infty x[k]\delta[n-k]$$
and for a linear system, given $h_k[k] = T[\delta[n-k]]$,
$$y[n] = \sum_{k=-\infty}^\infty x[k]h_k[n]$$
and if it is also time-invariant, $h_k[n] = h[n-k]$ where $h[n]=T[\delta[n]]$, so we can completely characterize an LTI system using its impulse response:
\begin{equation}\label{eq:lti}
       y[n] = \sum_{k=-\infty}^\infty x[k]h[n-k]
\end{equation}
This is the output of the system when the input is a unit sample.
\begin{equation}
	   h[n] = T\{\delta[n]\}
\end{equation}

\subsection{Convolution}
Equation \ref{eq:lti} is the DT convolution sum
\begin{equation}
    y[n]= x[n]*h[n] =(x * h)[n] \equiv \sum_{k=-\infty}^\infty x[k]h[n-k]
\end{equation}
Convolution can be thought of as the superposition of scaled and shifted impulse responses, or (for a fixed $n$) as the sum of products between $x[k]$ and a flipped and shifted $h[k]$ (over all $k$).

\subsubsection{Worked Example}
Consider a window function (chosen for finite support)
\begin{equation*}
    h[n]  = [n]-u[n-N]  = \begin{cases*}
        1 & $0\le n \le N-1$\\
        0 & otherwise
    \end{cases*}
\end{equation*}
and a decaying exponential ($a<1$) (chosen for boundedness)
\begin{equation*}
    x[n]  = a^nu[n] = \begin{cases*}
        a^n & $n\ge 0$\\
        0 & $n<0$
    \end{cases*}
\end{equation*}
Visualizing this situation as a flip and shift of the window, we can better understand that
\[x[n]*h[n]  = \begin{cases*}
    0 & if $n<0$\\
    \sum_{k=0}^n a^k & if $n\le N-1$\\
    \sum_{k=n-(N-1)}^n a^k & otherwise\\
\end{cases*}\]
Recall that $$S_n = \sum_{k=0}^n a^k =\frac{1-a^{n+1}}{1-a} $$ and so
\[x[n]*h[n]  = \begin{cases*}
    0 & if $n<0$\\
    \frac{1-a^{n+1}}{1-a} & if $n\le N-1$\\
    a^{n-N+1}\frac{1-a^{n+1}}{1-a} & otherwise\\
\end{cases*}\]

\subsection{Properties}
\subsubsection{Commutativity}
The order of convolution does not matter, swapping the ``input'' and ``output'' signals will not change the result.
\begin{equation}
    x[n]*h[n] = h[n]*x[n]
\end{equation}
We can prove this relation by substituting $m=n-k$ as below
\begin{align*}
    x[n]*h[n] =
    \sum_{k=-\infty}^\infty x[k]h[n-k] =
    \sum_{m=-\infty}^\infty x[m-n]h[m] =
    h[n]*x[n]
\end{align*}

\subsubsection{Distributivity}
\begin{equation}
    x[n]*(h_1[n]+h_2[n]) = x[n]*h_1[n]+x[n]*h_2[n]
\end{equation}

\subsubsection{Associativity}
Reordering and combining ``input'' signal and LTI systems will yield the same result.
\begin{equation}
    (x_1[n]*x_2[n])*x_3[n]=x_1[n]*(x_2[n]+x_3[n])
\end{equation}

\subsubsection{Impulse Response Test for BIBO Stability}
An LTI system is BIBO stable if and only if the impulse response is absolutely summable.
\begin{equation}
    B_h=\sum_{k=-\infty}^\infty |h[k]|<\infty
\end{equation}

% Lecture 3
Given an absolutely summable $h$ and $\forall n, |x[n]|\le B_x\le\infty$
\begin{align*}
	y[n] &= \sum_{k=-\infty}^\infty h[k]x[n-k]\\
	|y[n]| &= \left|\sum_{k=-\infty}^\infty h[k]x[n-k]\right|
	    \le \sum_{k=-\infty}^\infty |h[k]x[n-k]|
	    \le \sum_{k=-\infty}^\infty |h[k]||x[n-k]|\\
	    &\le \sum_{k=-\infty}^\infty |h[k]|B_x
	    = B_x\sum_{k=-\infty}^\infty |h[k]| = B_xB_h \le \infty
\end{align*}

To establish that BIBO stability implies absolute summability, we prove the contrapositive, that $h[n]$ not absolutely summable implies that there is no BIBO stability.

Given $\sum_{k=-\infty}^\infty |h[n]|=\infty$, consider bounded input
\[
    x[n] = \begin{cases*}
        \frac{h^*[-n]}{|h[-n]|} & if $h[n]\ne0$\\
        0 & otherwise
    \end{cases*}
\]
which has $|x[n]| \le 1$. Evaluating
\begin{align*}
	y[0] &= \sum_{k=-\infty}^\infty x[-k]h[k]\\
	     &= \sum_{k=-\infty}^\infty \frac{h^*[k]}{|h[k]|}h[k]
	      = \sum_{k=-\infty}^\infty |h[k]| = \infty
\end{align*}
and so an $h[n]$ that is not absolutely summable will not yield a BIBO stable system.

\subsubsection{Impulse Response Test for Causality}
For an LTI system,
\[
    y[n] = \sum_{k=-\infty}^\infty x[n-k]h[k]
\]
we want to zero out $k<0$ to prevent $x[n-k]$ to reach into the future. Therefore, an LTI system is causal when $\forall n<0$
\begin{equation}
    h[n]=0
\end{equation}


%\section{Difference Equations, Fourier representations, and the DTFT}
\subsection{Impulse Response Classifications}
A \textbf{finite impulse response system (FIR)} has an impulse response with only finitely many nonzero samples, and $h[n]=0$ outside of that finite impulse. All FIR systems are BIBO stable.
Any FIR can be computed non-recursively, but some can also be computed recursively. A sufficiently long delay can make a non-causal FIR causal.

A \textbf{infinite impulse response system (IIR)} is the opposite, a $h[n]\ne0$ for infinitely many indices. Some IIR systems are BIBO stable.


\subsection{Inverse Systems}
If an LTI system of impulse response $h[n]$ and it has an inverse $h_i[n]$, then
\[ h[n]*h_i[n]  = h_i[n] * h[n] = \delta[n]\]
For instance, the accumulator system ($h[n] = u[n]$, analogous to integration) and the backward difference system ($h[n] = \delta[n]-\delta[n-1]$, analogous to differentiation) are inverse systems.

\subsection{Difference Equations}
The general difference equation describes an important class of LTI systems. We find sequences of constants $a_k,b_m$ such that
\[
    \sum_{k=0}^N a_ky[n-k] = \sum_{m=0}^M b_m x[n-m]
\]
This is an $N^{th}$ order linear constant-coefficient difference equation. If previous output values are referenced, there is a recursive representation.

Solving for the output yields
\begin{equation}\label{eq:difference-solved}
	y[n] = -\sum_{k=1^N}\frac{a_k}{a_0}y[n-k] + \sum_{k=0}^M \frac{b_k}{a_0}x[n-k]
\end{equation}
and we note that in the $N=0$ case, this is equivalent to convolving with \[
    h[n] = \begin{cases*}
        \frac{b_n}{a_0} & $0\le n < M$\\
        0 & else
    \end{cases*}
\]

\subsubsection{Examples}
The accumulator function can be written as
\begin{align}
	y[n] &= \sum_{k=\infty}^n x[k] = x[n] + \sum_{k=\infty}^{n-1} x[k]\\
	     &= x[n] + y[n-1]\\
	y[n]-y[n-1] &= x[n]
\end{align}
so the accumulator can be represented recursively as an adder that adds $x[n]$ to the time-delayed $y[n-1]$.

An FIR moving average with $M_1=0$ can be represented non-recursively as
\[
    y[n] = \frac{1}{M_2+1}\sum_{k=0}^{M_2}x[n-k]
\]
but also we can factor the impulse response
\[
    h[n] = \frac{1}{M_2+1}\left(u[n]-u[n-M_2-1]\right)
         = \frac{1}{M_2+1}\left(\delta[n]-\delta[n-M_2-1]\right)*u[n]
\]
which can be implemented as a scaling, a delayed subtraction, and a recursive accumulation.

\subsubsection{Solutions to Difference Equations}
For a given input $x_p[n]$ with possible solution $y_p[n]$, the general form of solutions to the difference equation take the form
\[
    y[n] = y_p[n] + y_h[n]
\]
where $y_h[n]$ is a homogeneous solution satisfying the homogeneous difference equation
\[
    \sum_{k=0}^N a_k y_h[n-k]=0
\]
and for $N\ge1$, in order to discern the specific solution $y[n]$, $N$ initial conditions are necessary.

Just as $\dv{t}(e^{at})=a\left(e^{at}\right)$ ($e^{at}$ is an eigenfunction under differentiation), $z^n$ is the analogous eigenfunction under DT delays. A delay \[
    D\{x[n]\} = x[n-1]
\] applied $k$ times to an exponential $z^n$ yields \[
    D^K\{z^n\} =z^{-k}z^n
\]
and we write a polynomial in delays to represent the homogeneous difference equation:
\[
    (a_0+a_1D+a_2D^2+\cdots+a_ND^N)y_h[n]=0
\]
using $z^n$ as the trial sequence
\[
    (a_0+a_1z^{-1}+a_2z^{-2}+\cdots+a_Nz^{-N})z^n = z^n\sum_{k=0}^N a_kz^{-k} =0
\]
Labeling the summation as $A(z)$, we call the $z_m$ roots of $A(z)$ \textbf{characteristic roots}, which identify a solution of the form $z^n_m$.

For distinct roots, we can linearly combine all the solutions (scaling by undetermined coefficients $A_m$)
\[
    y_h[n] = \sum_{m=1}^N A_mz_m^n
\]
which require $N$ initial conditions (such as $y[-1], y[-2], \cdots, y[-N]$). When the system is expressed by a recursive difference equation, whether the system is linear, time-invariant, and causal all depend on those auxiliary conditions. \textbf{Initial rest} is one such choice (used for causal filter design) is to let
\begin{align}
	\forall n<n_0, x[n] = 0 \implies \forall n<n_0, y[n] = 0
\end{align}

\section{Fourier Representations}
An LTI system changes an input signal's complex amplitude by a complex scale factor (the \textbf{frequency response} $H(e^{j\omega})$), but not its fundamental frequency. Consider convolving an input $x[n]=e^{j \omega n}$ and defining $m=n-k$
\begin{align*}
   y[n] &= \sum_{k=-\infty}^\infty e^{j\omega k}h[n-k]\\
        &= \sum_{m=-\infty}^\infty e^{j\omega {m + n}}h[m]\\
        &= e^{j\omega n}\sum_{m=-\infty}^\infty e^{j\omega {m}}h[m]\\
        &= H(e^{j\omega})e^{j\omega n}\\
        &= \left|H(e^{j\omega})\right|e^{j(\omega n + \angle H(e^{j\omega})}
\end{align*}
where we express the complex frequency response in rectangular and polar forms
\[
    H(e^{j\omega}) = H_{Re}(e^{j\omega}) + H_{Im}(e^{j\omega}) = \left|H(e^{j\omega})\right|e^{\angle H(e^{j\omega})}
\]

\subsection{Real Impulse Responses}
For real $h[n]$, $h[n]=h^*[n]$
\[
     H(e^{j\omega}) = \sum_{n=-\infty}^\infty h[n]e^{-j\omega n}
\]
and then taking the complex conjugate of the frequency response,
\[
    H^*(e^{j\omega})
                    = \sum_{n=-\infty}^\infty h^*[n]e^{j\omega n}
                    = \sum_{n=-\infty}^\infty h[n]e^{-j(-\omega) n}
                    = H(e^{-j\omega})
\]
so the frequency response at opposite frequencies are complex conjugates.


\subsubsection{Example: Ideal Delay}
$y[n] = x[n-n_d]$ has \[
    H(e^{j\omega}) = e^{j\omega n_d}
\] so the magnitude is always $1$ and the phase shift is linear.
